   155→  tgts.forEach(t=>{
   156→    if(t.base){applyDamage(u,t,a,{});return}
   157→    let dmg=a,savHit=false;
   158→    if(sav&&!isCrit){const tt=t.traits||[]; // SAVAGE: xN only vs matching traits (crit takes precedence; metal rule still applies inside applyDamage)
